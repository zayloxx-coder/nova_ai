def obtenir_session_chat(reinitialiser=False):
    global client, chat_session
    if CLE_API == "COLLE_TA_CLE_API_ICI" or not CLE_API.strip():
        return None

    if client is None:
        client = genai.Client(api_key=CLE_API)

    if chat_session is None or reinitialiser:
        chat_session = client.chats.create(
            model=MODELE,
            config=types.GenerateContentConfig(
                system_instruction=INSTRUCTIONS_IA
            )
        )
    return chat_session

@app.route("/")
def accueil():
    return HTML_PAGE.replace("__NOM_IA__", html.escape(NOM_IA))

@app.route("/chat", methods=["POST"])
def chat():
    donnees = request.get_json()
    message_utilisateur = donnees.get("message", "")

    chat_actuel = obtenir_session_chat()
    if chat_actuel is None:
        return jsonify({
            "reponse": "⚠️ Tu n'as pas encore mis ta clé API ! Ouvre mon_ia.py et remplace COLLE_TA_CLE_API_ICI par ta clé."
        })

    try:
        reponse = chat_actuel.send_message(message_utilisateur)
        return jsonify({"reponse": reponse.text})
    except Exception as e:
        return jsonify({"reponse": f"❌ Erreur : {str(e)}"})

@app.route("/reset", methods=["POST"])
def reset():
    obtenir_session_chat(reinitialiser=True)
    return jsonify({"status": "ok"})

def ouvrir_navigateur():
    webbrowser.open("http://127.0.0.1:5000")
